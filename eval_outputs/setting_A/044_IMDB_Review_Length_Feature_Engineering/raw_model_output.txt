import numpy as np
from tensorflow import keras
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score, classification_report
from sklearn.preprocessing import StandardScaler

# Set random seeds for reproducibility
np.random.seed(42)

# Load IMDB dataset
# num_words limits vocabulary to top 10000 most frequent words
num_words = 10000
(x_train, y_train), (x_test, y_test) = keras.datasets.imdb.load_data(num_words=num_words)

print(f"Training samples: {len(x_train)}")
print(f"Test samples: {len(x_test)}")
print(f"Example review (word indices): {x_train[0][:20]}...")  # First 20 words
print(f"Example label: {y_train[0]}")

# Feature 1: Create review length feature
train_lengths = np.array([len(review) for review in x_train]).reshape(-1, 1)
test_lengths = np.array([len(review) for review in x_test]).reshape(-1, 1)

print(f"\nReview length statistics (train):")
print(f"  Mean: {train_lengths.mean():.2f}")
print(f"  Std: {train_lengths.std():.2f}")
print(f"  Min: {train_lengths.min()}")
print(f"  Max: {train_lengths.max()}")

# Feature 2: Create bag-of-words representation
# Convert sequences to binary bag-of-words vectors
def sequences_to_bow(sequences, num_words):
    """
    Convert sequences of word indices to binary bag-of-words vectors.
    
    Args:
        sequences: List of sequences (each sequence is a list of word indices)
        num_words: Vocabulary size
    
    Returns:
        Binary matrix of shape (len(sequences), num_words)
    """
    bow_matrix = np.zeros((len(sequences), num_words))
    for i, sequence in enumerate(sequences):
        for word_idx in sequence:
            if word_idx < num_words:
                bow_matrix[i, word_idx] = 1
    return bow_matrix

print("\nCreating bag-of-words representations...")
train_bow = sequences_to_bow(x_train, num_words)
test_bow = sequences_to_bow(x_test, num_words)

print(f"Bag-of-words shape (train): {train_bow.shape}")
print(f"Bag-of-words shape (test): {test_bow.shape}")

# Standardize the length feature
scaler = StandardScaler()
train_lengths_scaled = scaler.fit_transform(train_lengths)
test_lengths_scaled = scaler.transform(test_lengths)

# Model 1: Logistic Regression WITHOUT length feature (bag-of-words only)
print("\n" + "="*60)
print("Model 1: Logistic Regression with Bag-of-Words ONLY")
print("="*60)

model_bow_only = LogisticRegression(max_iter=100, random_state=42, solver='lbfgs')
model_bow_only.fit(train_bow, y_train)

y_pred_bow_only = model_bow_only.predict(test_bow)
accuracy_bow_only = accuracy_score(y_test, y_pred_bow_only)

print(f"\nAccuracy (BoW only): {accuracy_bow_only:.4f}")
print("\nClassification Report (BoW only):")
print(classification_report(y_test, y_pred_bow_only, target_names=['Negative', 'Positive']))

# Model 2: Logistic Regression WITH length feature (bag-of-words + length)
print("\n" + "="*60)
print("Model 2: Logistic Regression with Bag-of-Words + Length Feature")
print("="*60)

# Combine bag-of-words with length feature
train_combined = np.hstack([train_bow, train_lengths_scaled])
test_combined = np.hstack([test_bow, test_lengths_scaled])

print(f"Combined features shape (train): {train_combined.shape}")
print(f"Combined features shape (test): {test_combined.shape}")

model_combined = LogisticRegression(max_iter=100, random_state=42, solver='lbfgs')
model_combined.fit(train_combined, y_train)

y_pred_combined = model_combined.predict(test_combined)
accuracy_combined = accuracy_score(y_test, y_pred_combined)

print(f"\nAccuracy (BoW + Length): {accuracy_combined:.4f}")
print("\nClassification Report (BoW + Length):")
print(classification_report(y_test, y_pred_combined, target_names=['Negative', 'Positive']))

# Compare performance
print("\n" + "="*60)
print("Performance Comparison")
print("="*60)
print(f"Accuracy (BoW only):        {accuracy_bow_only:.4f}")
print(f"Accuracy (BoW + Length):    {accuracy_combined:.4f}")
print(f"Improvement:                {accuracy_combined - accuracy_bow_only:.4f}")

if accuracy_combined > accuracy_bow_only:
    print("\nConclusion: Adding the length feature IMPROVED performance.")
elif accuracy_combined < accuracy_bow_only:
    print("\nConclusion: Adding the length feature DECREASED performance.")
else:
    print("\nConclusion: Adding the length feature had NO EFFECT on performance.")

# Additional analysis: Check correlation between review length and sentiment
positive_lengths = train_lengths[y_train == 1]
negative_lengths = train_lengths[y_train == 0]

print("\n" + "="*60)
print("Review Length Analysis by Sentiment")
print("="*60)
print(f"Average length (Positive reviews): {positive_lengths.mean():.2f}")
print(f"Average length (Negative reviews): {negative_lengths.mean():.2f}")
print(f"Difference: {abs(positive_lengths.mean() - negative_lengths.mean()):.2f}")
