import numpy as np
import matplotlib.pyplot as plt
from tensorflow import keras
from collections import Counter

# Set random seed for reproducibility
np.random.seed(42)

# Load the Reuters newswire dataset
# num_words=None means we load all words (though we only need labels here)
(x_train, y_train), (x_test, y_test) = keras.datasets.reuters.load_data(num_words=None)

# Combine train and test labels to get all topics
all_labels = np.concatenate([y_train, y_test])

# Count occurrences of each topic
topic_counts = Counter(all_labels)

# Get the top 10 most common topics
top_10_topics = topic_counts.most_common(10)

# Extract topic IDs and their counts
topic_ids = [topic for topic, count in top_10_topics]
counts = [count for topic, count in top_10_topics]

# Calculate the total number of articles
total_articles = len(all_labels)

# Calculate the frequency of the most common topic
most_frequent_topic_count = counts[0]
most_frequent_topic_frequency = most_frequent_topic_count / total_articles

# Verify that the most frequent topic appears in at least 15% of articles
assert most_frequent_topic_frequency >= 0.15, f"Most frequent topic appears in {most_frequent_topic_frequency:.2%} of articles, which is less than 15%"

print(f"Total number of articles: {total_articles}")
print(f"Most frequent topic (ID {topic_ids[0]}): {most_frequent_topic_count} occurrences ({most_frequent_topic_frequency:.2%})")
print(f"\nTop 10 topics:")
for i, (topic_id, count) in enumerate(top_10_topics, 1):
    print(f"{i}. Topic {topic_id}: {count} articles ({count/total_articles:.2%})")

# Plot the top 10 topics as a bar chart
plt.figure(figsize=(10, 6))
plt.bar(range(len(topic_ids)), counts, color='steelblue', edgecolor='black')
plt.xlabel('Topic ID', fontsize=12)
plt.ylabel('Number of Articles', fontsize=12)
plt.title('Top 10 Most Frequent Topics in Reuters Newswire Dataset', fontsize=14, fontweight='bold')
plt.xticks(range(len(topic_ids)), topic_ids)
plt.grid(axis='y', alpha=0.3, linestyle='--')
plt.tight_layout()
plt.show()

print(f"\nVerification passed: The most frequent topic appears in {most_frequent_topic_frequency:.2%} of articles (>= 15%).")
