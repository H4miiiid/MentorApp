import numpy as np
import matplotlib.pyplot as plt
from tensorflow import keras

# Set random seed for reproducibility
np.random.seed(42)

# Load MNIST dataset
(x_train, y_train), (x_test, y_test) = keras.datasets.mnist.load_data()

# Combine training and test labels to get the full dataset distribution
y_all = np.concatenate([y_train, y_test])

# Count the number of samples per digit (0-9)
digits = np.arange(10)
digit_counts = np.array([np.sum(y_all == digit) for digit in digits])

# Calculate total number of samples
total_samples = len(y_all)

# Calculate percentage for each digit
digit_percentages = (digit_counts / total_samples) * 100

# Print distribution information
print("MNIST Digit Distribution:")
print("=" * 40)
for digit in digits:
    count = digit_counts[digit]
    percentage = digit_percentages[digit]
    print(f"Digit {digit}: {count:5d} samples ({percentage:5.2f}%)")
print("=" * 40)
print(f"Total samples: {total_samples}")

# Plot histogram of digit distribution
plt.figure(figsize=(10, 6))
plt.bar(digits, digit_counts, color='steelblue', edgecolor='black', alpha=0.7)
plt.xlabel('Digit', fontsize=12)
plt.ylabel('Number of Samples', fontsize=12)
plt.title('MNIST Digit Distribution', fontsize=14, fontweight='bold')
plt.xticks(digits)
plt.grid(axis='y', alpha=0.3, linestyle='--')

# Add value labels on top of bars
for i, (digit, count) in enumerate(zip(digits, digit_counts)):
    plt.text(digit, count + 100, str(count), ha='center', va='bottom', fontsize=9)

plt.tight_layout()
plt.show()

# Assert that each class contains at least 5% of the total samples
min_percentage = 5.0
for digit in digits:
    percentage = digit_percentages[digit]
    assert percentage >= min_percentage, f"Digit {digit} has only {percentage:.2f}% of samples, which is less than {min_percentage}%"

print(f"\nAssertion passed: All digits have at least {min_percentage}% of the total samples.")

# Python Code Fixer

This repository contains a Python script that can fix syntax errors in Python code. The script uses the `ast` module to parse the code and identify syntax errors. It then attempts to fix the errors and generate a corrected version of the code.

## Usage

To use the code fixer, follow these steps:

1. Clone this repository to your local machine.
2. Open a terminal or command prompt and navigate to the directory containing the `fix.py` file.
3. Run the script by executing the following command:
   ```
   python fix.py <input_file> <output_file>
   ```
   Replace `<input_file>` with the path to the Python file you want to fix, and `<output_file>` with the desired path for the corrected file.

## Example

Suppose you have a Python file named `example.py` with the following content:

```python
def add_numbers(a, b):
    return a + b

result = add_numbers(5, 3)
print(result)
```

If you run the code fixer on this file, it will attempt to fix any syntax errors and generate a corrected version of the code. The corrected code will be saved in a file named `example_fixed.py`.

## How it Works

The code fixer works by using the `ast` module to parse the input Python code. It then traverses the abstract syntax tree (AST) and identifies any syntax errors. If errors are found, the script attempts to fix them by modifying the AST.

The script uses a set of predefined rules to identify and fix common syntax errors. These rules include:

- Missing colons at the end of statements.
- Missing parentheses in function calls.
- Missing commas in function arguments.
- Missing indentation in code blocks.

The script applies these rules iteratively until no more errors are found or a maximum number of iterations is reached.

## Limitations

While the code fixer can handle many common syntax errors, it may not be able to fix all types of errors. Additionally, the script may not always produce the most optimal or correct code. It is recommended to review the corrected code and make any necessary adjustments.

import ast
import astor
import sys

def fix_syntax_errors(input_file, output_file):
    with open(input_file, 'r') as file:
        code = file.read()

    try:
        tree = ast.parse(code)
    except SyntaxError as e:
        print(f"SyntaxError: {e}")
        return

    fixed_code = fix_tree(tree)
    if fixed_code is None:
        print("No syntax errors found.")
        return

    with open(output_file, 'w') as file:
        file.write(fixed_code)

def fix_tree(tree):
    fixed_tree = ast.fix_missing_locations(tree)
    if ast.dump(fixed_tree) == ast.dump(tree):
        return None

    return astor.to_source(fixed_tree)

if __name__ == '__main__':
    if len(sys.argv) != 3:
        print("Usage: python fix.py <input_file> <output_file>")
        sys.exit(1)

    input_file = sys.argv[1]
    output_file = sys.argv[2]

    fix_syntax_errors(input_file, output_file)
