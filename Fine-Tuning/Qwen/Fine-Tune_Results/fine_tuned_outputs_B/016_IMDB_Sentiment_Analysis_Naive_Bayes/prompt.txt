import numpy as np
from tensorflow import keras
from sklearn.naive_bayes import MultinomialNB
from sklearn.metrics import accuracy_score

# Set random seed for reproducibility
np.random.seed(42)

# Load IMDB dataset from keras
# num_words limits vocabulary to the top 10000 most frequent words
num_words = 10000
(x_train, y_train), (x_test, y_test) = keras.datasets.imdb.load_data(num_words=num_words)

print(f"Training samples: {len(x_train)}")
print(f"Test samples: {len(x_test)}")
print(f"Example review (encoded): {x_train[0][:20]}...")  # First 20 word indices
print(f"Example label: {y_train[0]} (0=negative, 1=positive)\n")

# Convert sequences to bag-of-words vectors
# Each review is represented as a vector of word counts
def sequences_to_bow(sequences, num_words):
    """
    Convert sequences of word indices to bag-of-words vectors.
    
    Args:
        sequences: List of sequences (lists of word indices)
        num_words: Vocabulary size
    
    Returns:
        numpy array of shape (num_sequences, num_words) with word counts
    """
    bow_matrix = np.zeros((len(sequences), num_words), dtype=np.int32)
    for i, seq in enumerate(sequences):
        for word_idx in seq:
            if word_idx < num_words:
                bow_matrix[i, word_idx] += 1
    return bow_matrix

# Convert training and test data to bag-of-words
print("Converting sequences to bag-of-words vectors...")
X_train_bow = sequences_to_bow(x_train, num_words)
X_test_bow = sequences_to_bow(x_test, num_words)

print(f"Training BOW shape: {X_train_bow.shape}")
print(f"Test BOW shape: {X_test_bow.shape}\n")

# Train Multinomial Naive Bayes classifier
print("Training Multinomial Naive Bayes classifier...")
clf = MultinomialNB(alpha=1.0)
clf.fit(X_train_bow, y_train)

# Make predictions
y_train_pred = clf.predict(X_train_bow)
y_test_pred = clf.predict(X_test_bow)

# Calculate and report accuracy
train_accuracy = accuracy_score(y_train, y_train_pred)
test_accuracy = accuracy_score(y_test, y_test_pred)

print(f"Training Accuracy: {train_accuracy:.4f}")
print(f"Test Accuracy: {test_accuracy:.4f}\n")

# Get word index mapping to decode reviews
word_index = keras.datasets.imdb.get_word_index()
reverse_word_index = {value: key for key, value in word_index.items()}

def decode_review(encoded_review):
    """
    Decode an encoded review back to text.
    
    Args:
        encoded_review: List of word indices
    
    Returns:
        Decoded review string
    """
    # Note: indices are offset by 3 (0=padding, 1=start, 2=unknown)
    return ' '.join([reverse_word_index[i - 3] for i in encoded_review])
